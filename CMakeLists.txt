cmake_minimum_required(VERSION 3.22)
project(xrobot_cpp VERSION 0.1.0 LANGUAGES CXX)
add_link_options(-Wl,--disable-new-dtags)

option(XROBOT_CPP_WITH_XR "Build the native XRoboToolkit reader" OFF)
option(XROBOT_CPP_WITH_PLACO "Build the Placo C++ QP inverse-kinematics backend" OFF)
set(XROBOT_CPP_PLACO_PREFIX "/home/lyh/nero/conda_envs/xrobot-nero/lib/python3.10/site-packages/cmeel.prefix" CACHE PATH "Prefix containing Placo's Pinocchio/QP C++ dependencies")
set(XROBOT_CPP_MUJOCO_ROOT "/home/lyh/nero/piper_dual_arm_teleop/third_party/mujoco-3.3.2-linux-x86_64" CACHE PATH "MuJoCo SDK root containing include/mujoco and lib/libmujoco.so")
option(XROBOT_CPP_WITH_MUJOCO "Build the headless NERO MuJoCo physics backend" ON)
option(XROBOT_CPP_WITH_MUJOCO_VIEWER "Build the GLFW OpenGL MuJoCo viewer" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_BUILD_RPATH "/opt/ros/jazzy/lib;/opt/ros/jazzy/opt/sdformat_vendor/lib")

find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(kdl_parser REQUIRED PATHS /opt/ros/jazzy/share/kdl_parser/cmake NO_DEFAULT_PATH)
find_package(urdf REQUIRED PATHS /opt/ros/jazzy/share/urdf/cmake NO_DEFAULT_PATH)

add_library(xrobot_cpp_core
  src/control_core.cpp
  src/dynamics_controller.cpp
  src/nero_can.cpp
  src/nero_safety.cpp
  src/session_recorder.cpp
)
target_include_directories(xrobot_cpp_core PUBLIC include)
target_link_libraries(xrobot_cpp_core PUBLIC Eigen3::Eigen kdl_parser::kdl_parser urdf::urdf)
target_compile_options(xrobot_cpp_core PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(XROBOT_CPP_WITH_PLACO)
  list(PREPEND CMAKE_PREFIX_PATH "${XROBOT_CPP_PLACO_PREFIX}")
  find_package(pinocchio CONFIG REQUIRED)
  find_package(eiquadprog CONFIG REQUIRED)
  find_package(jsoncpp CONFIG REQUIRED)
  set(PLACO_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/placo EXCLUDE_FROM_ALL)
  target_sources(xrobot_cpp_core PRIVATE src/placo_ik.cpp)
  target_link_libraries(xrobot_cpp_core PUBLIC libplaco)
  target_compile_definitions(xrobot_cpp_core PUBLIC XROBOT_CPP_WITH_PLACO=1)
  set_property(TARGET xrobot_cpp_core APPEND PROPERTY BUILD_RPATH "${XROBOT_CPP_PLACO_PREFIX}/lib")
endif()

if(XROBOT_CPP_WITH_MUJOCO)
  find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h PATHS "${XROBOT_CPP_MUJOCO_ROOT}/include" NO_DEFAULT_PATH)
  find_library(MUJOCO_LIBRARY mujoco PATHS "${XROBOT_CPP_MUJOCO_ROOT}/lib" NO_DEFAULT_PATH)
  if(NOT MUJOCO_INCLUDE_DIR OR NOT MUJOCO_LIBRARY)
    message(FATAL_ERROR "MuJoCo was requested but not found under XROBOT_CPP_MUJOCO_ROOT=${XROBOT_CPP_MUJOCO_ROOT}")
  endif()
  target_sources(xrobot_cpp_core PRIVATE src/mujoco_simulation.cpp)
  target_include_directories(xrobot_cpp_core PRIVATE "${MUJOCO_INCLUDE_DIR}")
  target_link_libraries(xrobot_cpp_core PUBLIC "${MUJOCO_LIBRARY}")
  target_compile_definitions(xrobot_cpp_core PUBLIC XROBOT_CPP_WITH_MUJOCO=1)
  set_property(TARGET xrobot_cpp_core APPEND PROPERTY BUILD_RPATH "${XROBOT_CPP_MUJOCO_ROOT}/lib")
endif()

add_executable(xrobot_cpp src/main.cpp)
target_link_libraries(xrobot_cpp PRIVATE xrobot_cpp_core)
target_compile_options(xrobot_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(xrobot_cpp PRIVATE XROBOT_CPP_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

if(XROBOT_CPP_WITH_MUJOCO_VIEWER)
  if(NOT XROBOT_CPP_WITH_MUJOCO)
    message(FATAL_ERROR "XROBOT_CPP_WITH_MUJOCO_VIEWER requires XROBOT_CPP_WITH_MUJOCO")
  endif()
  set(OpenGL_GL_PREFERENCE "GLVND")
  find_package(OpenGL REQUIRED)
  find_package(glfw3 REQUIRED)
  target_include_directories(xrobot_cpp PRIVATE "${MUJOCO_INCLUDE_DIR}")
  target_link_libraries(xrobot_cpp PRIVATE OpenGL::GL glfw)
  target_compile_definitions(xrobot_cpp PRIVATE XROBOT_CPP_WITH_MUJOCO_VIEWER=1)
endif()

if(XROBOT_CPP_WITH_XR)
  set(XROBOT_CPP_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../xrobot/third_party/XRoboToolkit-PC-Service-Pybind")
  find_library(PXREA_SDK PXREARobotSDK PATHS "${XROBOT_CPP_SDK_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
  target_sources(xrobot_cpp PRIVATE src/xr_client.cpp)
  target_include_directories(xrobot_cpp PRIVATE "${XROBOT_CPP_SDK_ROOT}/include")
  target_link_libraries(xrobot_cpp PRIVATE "${PXREA_SDK}")
  target_compile_definitions(xrobot_cpp PRIVATE XROBOT_CPP_WITH_XR=1)
  set_target_properties(xrobot_cpp PROPERTIES BUILD_RPATH "${XROBOT_CPP_SDK_ROOT}/lib")
endif()

enable_testing()
add_executable(xrobot_cpp_core_tests tests/core_test.cpp)
target_link_libraries(xrobot_cpp_core_tests PRIVATE xrobot_cpp_core)
target_compile_options(xrobot_cpp_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME xrobot_cpp_core_tests COMMAND xrobot_cpp_core_tests)
if(XROBOT_CPP_WITH_MUJOCO)
  add_test(NAME xrobot_cpp_mujoco_model_parity COMMAND xrobot_cpp --mujoco-model-parity)
endif()
